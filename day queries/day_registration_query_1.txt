--#########COUNT IB AND MB

WITH report_dates AS (
    SELECT 
        (
        CURRENT_DATE::DATE AS today
)
SELECT 
    s.service::VARCHAR AS registration_details,
    -- Cumulative Total Count (<= today)
    SUM(CASE WHEN TRUNC(t.created_dt) <= d.today THEN 1 ELSE 0 END)::BIGINT AS total,
    -- today Count (Today's reported metric)
    SUM(CASE WHEN TRUNC(t.created_dt) = d.today THEN 1 ELSE 0 END)::BIGINT AS today
FROM (
    SELECT 'MB Count' AS service, 'MB' AS code
    UNION ALL 
    SELECT 'IB Count', 'IB'
) s
CROSS JOIN report_dates d
LEFT JOIN (
    -- MB Registrations
    SELECT 'MB' AS code, created_dt 
    FROM customerprofile.tl_cust_registration_dtls
    
    UNION ALL
    
    -- IB Registrations
    SELECT 'IB' AS code, created_dt 
    FROM customerprofile.tl_ib_cust_registration_dtls
) t ON s.code = t.code AND TRUNC(t.created_dt) <= d.today
GROUP BY s.service;
