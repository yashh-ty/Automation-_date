WITH report_dates AS (
    SELECT 
        (CURRENT_DATE - 1)::DATE AS yesterday,
        date_trunc('month', CURRENT_DATE)::DATE AS month_start,
        CURRENT_DATE::DATE AS today
),

cra_targets AS (
    SELECT 'NSDL' AS cra_code, 'NPS-NSDL account opened' AS service_name, 'NSDL Account Opened' AS type_name
    UNION ALL
    SELECT 'KFINTECH' AS cra_code, 'NPS-KFINTECH account opened' AS service_name, 'Kfintech Account Opened' AS type_name
)


-- 1. Fund Transfer Metrics
SELECT 
    'Fund transfer'::VARCHAR AS service,
    CASE
        WHEN t.schedule_pymt_id IS NOT NULL AND s.is_one_time=FALSE THEN 'SI RECURRING'
        WHEN t.schedule_pymt_id IS NOT NULL AND s.is_one_time=TRUE THEN 'SI-ONE TIME'
        WHEN t.trnx_mode='WITHINBOI' THEN 'Within bank'
        WHEN t.trnx_mode='SELF' THEN 'Self Transfer'
        ELSE t.trnx_mode
    END::VARCHAR AS type,
    SUM(CASE WHEN UPPER(t.trnx_status) = 'SUCCESS' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(t.trnx_status) = 'SUCCESS' THEN t.trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN UPPER(t.trnx_status) = 'FAILED' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(t.trnx_status) = 'FAILED' THEN t.trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM moneymovement.tl_transaction_dtls t
LEFT JOIN moneymovement.tl_scheduled_pymts s 
    ON t.schedule_pymt_id = s.schedule_pymt_id
CROSS JOIN report_dates d
WHERE TRUNC(t.created_dt) = d.yesterday
  AND NOT (UPPER(t.trnx_status) = 'FAILED' AND t.trnx_status_desc LIKE '%Maximum transfer limit%')
  AND t.trnx_mode <> 'UPIM'
  AND (
      (t.schedule_pymt_id IS NULL AND t.trnx_mode IN ('IMPS', 'NEFT', 'RTGS', 'SELF'))
      OR 
      (t.schedule_pymt_id IS NULL AND t.trnx_mode = 'WITHINBOI' AND t.source_sys NOT IN ('PPF', 'SSY', 'INSUR', 'BILPMT', 'NPS'))
      OR 
      (t.schedule_pymt_id IS NOT NULL AND s.schedule_pymt_id IS NOT NULL)
  )
GROUP BY 1, 2

UNION ALL

-- 2. UPI Merchant & Retail Transactions
SELECT 
    'UPI'::VARCHAR AS service,
    service AS type,
    SUM(CASE WHEN trnx_status = 'Paid' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN trnx_status = 'Paid' THEN payee_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN trnx_status <> 'Paid' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN trnx_status <> 'Paid' THEN payee_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM (
    SELECT 
        CASE 
            WHEN payee_code <> '0000' AND payee_ifsc LIKE '%BKID%' THEN 'To Merchant BOIAccount'
            WHEN payee_code <> '0000' AND payee_ifsc NOT LIKE '%BKID%' THEN 'To Merchant Non-BOI Account'
            WHEN payee_code = '0000' AND payee_ifsc LIKE '%BKID%' THEN 'To Retail BOI Account'
            WHEN payee_code = '0000' AND payee_ifsc NOT LIKE '%BKID%' THEN 'To Retail Non-BOI Account'
        END::VARCHAR AS service,
        trnx_status,
        payee_amount
    FROM moneymovement.tl_upi_trnx_dtls
    CROSS JOIN report_dates d
    WHERE txn_type = 'PAY' 
      AND TRUNC(created_dt) = d.yesterday
) upi_base
WHERE service IS NOT NULL
GROUP BY 1, 2

UNION ALL

-- 3. VPA Creation Metrics
SELECT 
    'UPI'::VARCHAR AS service,
    'VPA Creation'::VARCHAR AS type,
    SUM(CASE WHEN source_table = 'ADDRESS' THEN cnt ELSE 0 END)::BIGINT AS success_count,
    0::NUMERIC AS success_amount,
    SUM(CASE WHEN source_table = 'AUDIT' THEN cnt ELSE 0 END)::BIGINT AS failed_count,
    0::NUMERIC AS failed_amount
FROM (
    SELECT 'ADDRESS' AS source_table, COUNT(*) AS cnt
    FROM moneymovement.tl_payment_addresses 
    CROSS JOIN report_dates d
    WHERE migrated_rec = 'N' 
      AND TRUNC(created_dt) = d.yesterday

    UNION ALL

    SELECT 'AUDIT' AS source_table, COUNT(*) AS cnt
    FROM moneymovement.tl_upi_audit 
    CROSS JOIN report_dates d
    WHERE api = 'addPaymentAddress' 
      AND status IS NULL 
      AND TRUNC(created_dt) = d.yesterday
) vpa_data

UNION ALL

-- 4. Credit Card Payments
SELECT 
    'Credit-card Payment'::VARCHAR AS service,
    'CC-Payment'::VARCHAR AS type,
    SUM(CASE WHEN status = 'Success' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN status = 'Success' THEN payment_amt END), 0) / 100.0, 2)::NUMERIC AS success_amount,
    SUM(CASE WHEN status = 'Failure' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN status = 'Failure' THEN payment_amt END), 0) / 100.0, 2)::NUMERIC AS failed_amount
FROM cards.tl_cc_pymt_trnx_dtls
CROSS JOIN report_dates d
WHERE status IN ('Success', 'Failure')
  AND TRUNC(created_dt) = d.yesterday

UNION ALL

-- 5. SI Execution Metrics
SELECT 
    'SI Executed'::VARCHAR AS service,
    s_list.status_name::VARCHAR AS type,
    COALESCE(SUM(p.cnt), 0) ::BIGINT AS success_count,
    ROUND(COALESCE(SUM(p.amt), 0),0)::NUMERIC AS success_amount,
    0::BIGINT AS failed_count,
    0::NUMERIC AS failed_amount
FROM (
    SELECT 'ACTIVE' AS status_name
    UNION ALL SELECT 'COMPLETED'
    UNION ALL SELECT 'CANCELLED'
    UNION ALL SELECT 'PAUSED'
) s_list
LEFT JOIN (
    SELECT 
        UPPER(status) AS status_name,
        COUNT(*) AS cnt,
        SUM(trnx_amount) AS amt
    FROM moneymovement.tl_scheduled_pymts
    CROSS JOIN report_dates d
    WHERE TRUNC(created_dt) = d.yesterday
    GROUP BY UPPER(status)
) p ON s_list.status_name = p.status_name
GROUP BY 1, 2

UNION ALL

-- 6. Bill Desk BillPay

    'BillPay BillDesk'::VARCHAR AS service,
    'BPAY'::VARCHAR AS type,
    SUM(CASE WHEN UPPER(trnx_status) = 'SUCCESS' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(trnx_status) = 'SUCCESS' THEN debit_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN UPPER(trnx_status) = 'FAILED' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(trnx_status) = 'FAILED' THEN debit_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM moneymovement.tl_billpay_trnx_dtls
CROSS JOIN report_dates d
WHERE UPPER(trnx_status) IN ('SUCCESS', 'FAILED')
  AND TRUNC(created_dt) = d.yesterday

UNION ALL

-- 7. TD Contributions (FD / RD)
SELECT 
    'TD Contribution'::VARCHAR AS service,
    td_type::VARCHAR AS type,
    SUM(CASE WHEN UPPER(status) = 'SUCCESS' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(status) = 'SUCCESS' THEN dep_amount ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN UPPER(status) = 'FAILED' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(status) = 'FAILED' THEN dep_amount ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM investment.tl_td_trnx_dtls
CROSS JOIN report_dates d
WHERE td_type IN ('FD', 'RD')
  AND UPPER(status) IN ('SUCCESS', 'FAILED')
  AND TRUNC(created_dt) = d.yesterday
GROUP BY td_type

UNION ALL

-- 8. PPF & SSY Contributions
SELECT 
    (source_sys || ' Contribution')::VARCHAR AS service,
    source_sys::VARCHAR AS type,
    SUM(CASE WHEN UPPER(trnx_status) = 'SUCCESS' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(trnx_status) = 'SUCCESS' THEN trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN UPPER(trnx_status) = 'FAILED' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(trnx_status) = 'FAILED' THEN trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM moneymovement.tl_transaction_dtls
CROSS JOIN report_dates d
WHERE source_sys IN ('PPF', 'SSY')
  AND UPPER(trnx_status) IN ('SUCCESS', 'FAILED')
  AND TRUNC(created_dt) = d.yesterday
GROUP BY source_sys

UNION ALL

-- 9. NPS Contributions (NSDL & KFintech)
SELECT 
    CASE 
        WHEN a.pop_code = '5000424' THEN 'NPS-NSDL Contribution'
        WHEN a.pop_code = '1000504' THEN 'NPS-KFINTECH Contribution'
    END::VARCHAR AS service,
    CASE 
        WHEN a.pop_code = '5000424' AND a.nps_schedule_pymt_id IS NULL THEN 'NSDL Contribution'
        WHEN a.pop_code = '5000424' AND a.nps_schedule_pymt_id IS NOT NULL AND b.is_one_time = FALSE THEN 'NSDL-SI Recurring'
        WHEN a.pop_code = '1000504' THEN 'Kfintech trnx'
    END::VARCHAR AS type,
    SUM(CASE WHEN UPPER(a.status) = 'SUCCESS' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(a.status) = 'SUCCESS' THEN a.total_amt ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN UPPER(a.status) = 'FAILED' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(a.status) = 'FAILED' THEN a.total_amt ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM investment.tl_nps_trnx_dtls a
LEFT JOIN investment.tl_scheduled_nps_contribution_dtls b 
    ON a.nps_schedule_pymt_id = b.nps_schedule_pymt_id
CROSS JOIN report_dates d
WHERE TRUNC(a.created_dt) = d.yesterday
  AND UPPER(a.status) IN ('SUCCESS', 'FAILED')
  AND (
      (a.pop_code = '5000424' AND a.nps_schedule_pymt_id IS NULL)
      OR 
      (a.pop_code = '5000424' AND a.nps_schedule_pymt_id IS NOT NULL AND b.is_one_time = FALSE)
      OR 
      (a.pop_code = '1000504')
  )
GROUP BY 1, 2

UNION ALL

-- 10. Sovereign Gold Bonds (SGB)
SELECT 
    'SGB Contribution'::VARCHAR AS service,
    'SGB'::VARCHAR AS type,
    SUM(CASE WHEN source_table = 'SUCCESS' THEN cnt ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN source_table = 'SUCCESS' THEN total_amt ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN source_table = 'FAILED' THEN cnt ELSE 0 END)::BIGINT AS failed_count,
    0::NUMERIC AS failed_amount
FROM (
    SELECT 'SUCCESS' AS source_table, COUNT(*) AS cnt, SUM(trans_amt) AS total_amt
    FROM investment.tl_sgb_trnx_dtls
    CROSS JOIN report_dates d
    WHERE UPPER(status) = 'SUCCESS'
      AND TRUNC(created_dt) = d.yesterday

    UNION ALL

    SELECT 'FAILED' AS source_table, COUNT(*) AS cnt, 0 AS total_amt
    FROM investment.tl_sgb_trnx_error_dtls
    CROSS JOIN report_dates d
    WHERE TRUNC(created_dt) = d.yesterday
) sgb_data

UNION ALL

-- 11. Floating Rate Savings Bonds (FRSB)
SELECT 
    'FRSB Contribution'::VARCHAR AS service,
    'FRSB'::VARCHAR AS type,
    SUM(CASE WHEN source_table = 'SUCCESS' THEN cnt ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN source_table = 'SUCCESS' THEN total_amt ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN source_table = 'FAILED' THEN cnt ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN source_table = 'FAILED' THEN total_amt ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM (
    SELECT 'SUCCESS' AS source_table, COUNT(*) AS cnt, SUM(subs_amount) AS total_amt
    FROM investment.tl_frsb_trnx_dtls
    CROSS JOIN report_dates d
    WHERE UPPER(status) = 'SUCCESS'
      AND TRUNC(created_dt) = d.yesterday

    UNION ALL

    SELECT 'FAILED' AS source_table, COUNT(*) AS cnt, SUM(subscription_amt) AS total_amt
    FROM investment.tl_frsb_trnx_error_dtls
    CROSS JOIN report_dates d
    WHERE TRUNC(created_dt) = d.yesterday
) frsb_data

UNION ALL

-- 12. Mutual Fund Lumpsum & NFO
SELECT 
    d_cat.type_name::VARCHAR AS service,
    d_cat.service_name::VARCHAR AS type,
    COUNT(CASE WHEN t.fund_trnx_id IS NOT NULL THEN 1 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN t.fund_trnx_id IS NOT NULL THEN t.total_amt ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    COUNT(CASE WHEN t.fund_trnx_id IS NULL AND t.created_dt IS NOT NULL THEN 1 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN t.fund_trnx_id IS NULL AND t.created_dt IS NOT NULL THEN t.total_amt ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM (
    SELECT 'NFO' AS service_name, 'Multi Asset Allocation Fund' AS type_name, 'NFO' AS cat
    UNION ALL
    SELECT 'MF-LUMPSUM' AS service_name, 'MF-LUMSUM' AS type_name, 'LUMPSUM' AS cat
) d_cat
LEFT JOIN investment.tl_mf_transaction_dtls t
    ON (
        (d_cat.cat = 'NFO' AND UPPER(TRIM(t.isin_code)) IN ('INF761K01FW0', 'INF761K01FX8', 'INF761K01FY6'))
        OR
        (d_cat.cat = 'LUMPSUM' AND (UPPER(TRIM(t.isin_code)) NOT IN ('INF761K01FW0', 'INF761K01FX8', 'INF761K01FY6') OR t.isin_code IS NULL))
    )
    AND TRUNC(t.created_dt) = (SELECT yesterday FROM report_dates LIMIT 1)
GROUP BY d_cat.type_name, d_cat.service_name

UNION ALL

-- 16. Overdraft (OD) Contributions
SELECT 
    v.td_type::VARCHAR AS service,
    'OD Contribution'::VARCHAR AS type,
    SUM(CASE WHEN t.status = 'Success' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN t.status = 'Success' THEN t.od_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN t.status = 'Failed' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN t.status = 'Failed' THEN t.od_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM (VALUES ('FD'), ('RD')) AS v(td_type)
LEFT JOIN report_dates d ON 1=1
LEFT JOIN investment.tl_od_trnx_dtls t 
    ON t.td_type = v.td_type 
   AND TRUNC(t.created_dt) = d.yesterday 
GROUP BY v.td_type

UNION ALL

-- 18. Insurance 
SELECT 
    v.type::VARCHAR AS service,
    v.service::VARCHAR AS type,
    -- Success metrics
    COUNT(CASE 
        WHEN v.is_sudlife = FALSE AND t.policy_no IS NOT NULL THEN 1
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NULL THEN 1
    END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE 
        WHEN v.is_sudlife = FALSE AND t.policy_no IS NOT NULL THEN t.proposal_amt::NUMERIC
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NULL THEN s.tot_premium_with_tax::NUMERIC
        ELSE 0 
    END), 0), 0)::NUMERIC AS success_amount,
    -- Failed metrics
    COUNT(CASE 
        WHEN v.is_sudlife = FALSE AND t.created_dt IS NOT NULL AND t.policy_no IS NULL THEN 1
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NOT NULL THEN 1
    END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE 
        WHEN v.is_sudlife = FALSE AND t.created_dt IS NOT NULL AND t.policy_no IS NULL THEN t.proposal_amt::NUMERIC
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NOT NULL THEN s.tot_premium_with_tax::NUMERIC
        ELSE 0 
    END), 0), 0)::NUMERIC AS failed_amount
FROM (
    VALUES 
        ('Insurance Purchase Contribution', 'INSUR Reliance Health Premium', 'Reliance', 'health', FALSE),
        ('Insurance Purchase Contribution', 'INSUR Reliance Vehicle Premium', 'Reliance', 'vehicle', FALSE),
        ('Insurance Sudlife Purchase Contribution', 'SUDLIFE', NULL, NULL, TRUE)
) AS v(service, type, partner_name, prdct_type, is_sudlife)
LEFT JOIN report_dates d ON 1=1
-- Join standard insurance table
LEFT JOIN investment.tl_ins_purchase_dtls t 
    ON v.is_sudlife = FALSE
   AND t.partner_name = v.partner_name 
   AND t.prdct_type = v.prdct_type
   AND TRUNC(t.created_dt) = d.yesterday 
-- Join SUDLIFE insurance table
LEFT JOIN investment.tl_ins_sudlife_purchase_dtls s 
    ON v.is_sudlife = TRUE
   AND TRUNC(s.created_dt) = d.yesterday 
GROUP BY v.type, v.service


-- ###Kfintech and NSDl Account Opened
UNION ALL

SELECT 
    t.service_name::VARCHAR AS service,
    t.type_name::VARCHAR AS type,
    
    -- Success Metrics
    COALESCE(SUM(
        CASE 
            WHEN UPPER(TRIM(p.cra_name)) = t.cra_code 
             AND p.acct_added_by = 'A' 
             AND p.pran_no IS NOT NULL 
            THEN 1 ELSE 0 
        END
    ), 0)::BIGINT AS success_count,
    
    0::NUMERIC AS success_amount,
    
    -- Failed Metrics
    0::NUMERIC AS failed_count,
    
    0::NUMERIC AS failed_amount

FROM cra_targets t

LEFT JOIN investment.tl_nps_cust_profile p
CROSS JOIN report_dates d
    ON UPPER(TRIM(p.cra_name)) = t.cra_code
   AND TRUNC(p.created_dt)=d.yesterday
   AND UPPER(TRIM(p.channel))='MOB'
GROUP BY t.service_name, t.type_name, t.cra_code
ORDER BY service;
