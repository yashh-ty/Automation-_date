WITH report_dates AS (
    SELECT 
        (CURRENT_DATE - 1)::DATE AS yesterday,
        date_trunc('month', CURRENT_DATE)::DATE AS month_start,
        CURRENT_DATE::DATE AS today
),

cra_targets AS (
    SELECT 'NSDL' AS cra_code, 'NPS-NSDL account opened' AS service_name, 'NSDL Account Opened' AS type_name
    UNION ALL
    SELECT 'KFINTECH' AS cra_code, 'NPS-KFINTECH account opened' AS service_name, 'Kfintech Account Opened' AS type_name
)

-- 1. Fund Transfers (tl_transaction_dtls - Non-scheduled)
SELECT 
    'Fund transfer'::VARCHAR AS service,
    trnx_mode::VARCHAR AS type,
    SUM(CASE WHEN trnx_status = 'Success' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN trnx_status = 'Success' THEN trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN trnx_status = 'Failed' AND trnx_status_desc NOT LIKE '%Maximum transfer limit%' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN trnx_status = 'Failed' AND trnx_status_desc NOT LIKE '%Maximum transfer limit%' THEN trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM moneymovement.tl_transaction_dtls CROSS JOIN report_dates d
WHERE schedule_pymt_id IS NULL 
  AND TRUNC(created_dt) = d.yesterday 
  AND channel IN ('WEB', 'YLW')
  AND (
      trnx_mode IN ('IMPS', 'NEFT', 'RTGS', 'SELF')
      OR (trnx_mode = 'WITHINBOI' AND source_sys NOT IN ('PPF','SSY','INSUR','BILPMT','NPS'))
  )
GROUP BY trnx_mode

UNION ALL

-- 2. Fund Transfers (SI Scheduled)
SELECT 
    'Fund transfer'::VARCHAR AS service,
    (CASE WHEN b.is_one_time = TRUE THEN 'SI-ONE TIME' ELSE 'SI-RECURRING' END)::VARCHAR AS type,
    SUM(CASE WHEN a.trnx_status = 'Success' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN a.trnx_status = 'Success' THEN a.trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN a.trnx_status = 'Failed' AND a.trnx_status_desc NOT LIKE '%Maximum transfer limit%' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN a.trnx_status = 'Failed' AND a.trnx_status_desc NOT LIKE '%Maximum transfer limit%' THEN a.trnx_amount ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM moneymovement.tl_transaction_dtls a
JOIN moneymovement.tl_scheduled_pymts b ON a.schedule_pymt_id = b.schedule_pymt_id
CROSS JOIN report_dates d
WHERE a.schedule_pymt_id IS NOT NULL 
  AND TRUNC(a.created_dt) = d.yesterday 
  AND a.channel IN ('WEB', 'YLW')
GROUP BY b.is_one_time

UNION ALL

-- 3. Credit-Card Payments
SELECT 
    'Credit-card Payment'::VARCHAR AS service,
    'CC-Payment'::VARCHAR AS type,
    SUM(CASE WHEN LOWER(status) IN ('success', 'successful') THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN LOWER(status) IN ('success', 'successful') THEN payment_amt::NUMERIC ELSE 0 END), 0) / 100.0, 2)::NUMERIC AS success_amount,
    SUM(CASE WHEN LOWER(status) IN ('failure', 'failed') THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN LOWER(status) IN ('failure', 'failed') THEN payment_amt ::NUMERIC ELSE 0 END), 0) / 100.0, 2)::NUMERIC AS failed_amount
FROM cards.tl_cc_pymt_trnx_dtls CROSS JOIN report_dates d
WHERE TRUNC(created_dt) = d.yesterday AND channel IN ('WEB', 'YLW')

UNION ALL

    'OD Contribution'::VARCHAR AS service,
    v.td_type::VARCHAR AS type,
    SUM(CASE WHEN t.status = 'Success' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN t.status = 'Success' THEN t.od_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS success_amount,

SELECT 
    v.service::VARCHAR AS service,
    v.type::VARCHAR AS type,
    -- Success metrics
    COUNT(CASE 
        WHEN v.is_sudlife = FALSE AND t.policy_no IS NOT NULL THEN 1
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NULL THEN 1
    END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE 
        WHEN v.is_sudlife = FALSE AND t.policy_no IS NOT NULL THEN t.proposal_amt::NUMERIC
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NULL THEN s.tot_premium_with_tax::NUMERIC
        ELSE 0 
    END), 0), 0)::NUMERIC AS success_amount,
    -- Failed metrics
    COUNT(CASE 
        WHEN v.is_sudlife = FALSE AND t.created_dt IS NOT NULL AND t.policy_no IS NULL THEN 1
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NOT NULL THEN 1
    END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE 
        WHEN v.is_sudlife = FALSE AND t.created_dt IS NOT NULL AND t.policy_no IS NULL THEN t.proposal_amt::NUMERIC
        WHEN v.is_sudlife = TRUE  AND s.created_dt IS NOT NULL AND s.error_desc IS NOT NULL THEN s.tot_premium_with_tax::NUMERIC
        ELSE 0 
    END), 0), 0)::NUMERIC AS failed_amount
FROM (
    VALUES 
        ('Insurance Purchase Contribution', 'INSUR Reliance Health Premium', 'Reliance', 'health', FALSE),
        ('Insurance Purchase Contribution', 'INSUR Reliance Vehicle Premium', 'Reliance', 'vehicle', FALSE),
        ('Insurance Sudlife Purchase Contribution', 'SUDLIFE', NULL, NULL, TRUE)
) AS v(service, type, partner_name, prdct_type, is_sudlife)
LEFT JOIN report_dates d ON 1=1
-- Join standard insurance table
LEFT JOIN investment.tl_ins_purchase_dtls t 
    ON v.is_sudlife = FALSE
   AND t.partner_name = v.partner_name 
   AND t.prdct_type = v.prdct_type
   AND TRUNC(t.created_dt) = d.yesterday 
   AND t.channel IN ('WEB', 'YLW')
-- Join SUDLIFE insurance table
LEFT JOIN investment.tl_ins_sudlife_purchase_dtls s 
    ON v.is_sudlife = TRUE
   AND TRUNC(s.created_dt) = d.yesterday 
   AND s.channel IN ('WEB', 'YLW')
GROUP BY v.service, v.type

union all
SELECT 
    v.service::VARCHAR AS service,
    v.type::VARCHAR AS type,
    -- Success counts from customer profile tables
    COUNT(CASE 
        WHEN v.account_type = 'PPF' AND p.created_dt IS NOT NULL THEN 1
        WHEN v.account_type = 'SSY' AND s.created_dt IS NOT NULL THEN 1
    END)::BIGINT AS success_count,
    0::NUMERIC AS success_amount,
    -- Failed counts from registration error tables
    COUNT(CASE 
        WHEN v.account_type = 'PPF' AND pe.created_dt IS NOT NULL THEN 1
        WHEN v.account_type = 'SSY' AND se.created_dt IS NOT NULL THEN 1
    END)::BIGINT AS failed_count,
    0::NUMERIC AS failed_amount
FROM (
    VALUES 
        ('PPF Opened', 'PPF-Opened', 'PPF'),
        ('SSY Opened', 'SSY-Opened', 'SSY')
) AS v(service, type, account_type)
LEFT JOIN report_dates d ON 1=1
-- PPF Success Table
LEFT JOIN investment.tl_ppf_cust_profile p
    ON v.account_type = 'PPF'
   AND TRUNC(p.created_dt) = d.yesterday
   AND p.channel IN ('WEB', 'YLW')
-- PPF Error Table
LEFT JOIN investment.tl_ppf_regs_error_dtls pe
    ON v.account_type = 'PPF'
   AND TRUNC(pe.created_dt) = d.yesterday
   AND pe.channel IN ('WEB', 'YLW')
-- SSY Success Table
LEFT JOIN investment.tl_ssy_cust_profile s
    ON v.account_type = 'SSY'
   AND TRUNC(s.created_dt) = d.yesterday
   AND s.channel IN ('WEB', 'YLW')
-- SSY Error Table
LEFT JOIN investment.tl_ssy_regs_error_dtls se
    ON v.account_type = 'SSY'
   AND TRUNC(se.created_dt) = d.yesterday
   AND se.channel IN ('WEB', 'YLW')
GROUP BY v.service, v.type 

-- ###Kfintech and NSDl Account Opened
UNION ALL

SELECT 
    t.service_name::VARCHAR AS service,
    t.type_name::VARCHAR AS type,
    
    -- Success Metrics
    COALESCE(SUM(
        CASE 
            WHEN UPPER(TRIM(p.cra_name)) = t.cra_code 
             AND p.acct_added_by = 'A' 
             AND p.pran_no IS NOT NULL 
            THEN 1 ELSE 0 
        END
    ), 0)::BIGINT AS success_count,
    
    0::NUMERIC AS success_amount,
    
    -- Failed Metrics
    0::NUMERIC AS failed_count,
    
    0::NUMERIC AS failed_amount

FROM cra_targets t

LEFT JOIN investment.tl_nps_cust_profile p
CROSS JOIN report_dates d
    ON UPPER(TRIM(p.cra_name)) = t.cra_code
   AND TRUNC(p.created_dt)=d.yesterday
   AND UPPER(TRIM(p.channel)) in ('WEB', 'YLW')
GROUP BY t.service_name, t.type_name, t.cra_code
ORDER BY service;
