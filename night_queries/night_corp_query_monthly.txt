WITH report_dates AS (
    SELECT (CURRENT_DATE - 1)::DATE AS yesterday,
	date_trunc('month', CURRENT_DATE)::DATE AS month_start,
    CURRENT_DATE::DATE AS today
)

-- 1. Corporate Debit Fund Transfers
SELECT 
    'Fund transfer'::VARCHAR AS service,
    CASE 
        WHEN trnx_type = 'WITHINBOI' THEN 'Within bank'
        WHEN trnx_type = 'SELF' THEN 'Self Transfer'
        ELSE trnx_type 
    END::VARCHAR AS type,
    SUM(CASE WHEN LOWER(trnx_status) = 'success' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN LOWER(trnx_status) = 'success' THEN trnx_amount::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE 
        WHEN LOWER(trnx_status) = 'failed' 
         AND (trnx_type = 'BILL_PAYMENT' OR trnx_status_desc NOT LIKE '%Maximum transfer limit%') 
        THEN 1 ELSE 0 
    END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE 
        WHEN LOWER(trnx_status) = 'failed' 
         AND (trnx_type = 'BILL_PAYMENT' OR trnx_status_desc NOT LIKE '%Maximum transfer limit%') 
        THEN trnx_amount::NUMERIC ELSE 0 
    END), 0), 0)::NUMERIC AS failed_amount
FROM moneymovement.tl_corp_debit_trnx_dtls
CROSS JOIN report_dates d
WHERE trunc(created_dt) between month_start and yesterday
  AND (
      (schedule_pymt_id IS NULL AND trnx_type IN ('IMPS', 'NEFT', 'RTGS', 'SELF'))
      OR 
      (schedule_pymt_id IS NULL AND trnx_type = 'WITHINBOI' AND source_sys NOT IN ('PPF', 'SSY', 'INSUR', 'BILPMT', 'NPS'))
      OR 
      (trnx_type = 'BILL_PAYMENT')
  )
GROUP BY 1, 2

UNION ALL

-- 2. Corporate Bulk File Uploads
SELECT 
    'Corp Bulk'::VARCHAR AS service,
    trnx_type::VARCHAR AS type,
    SUM(CASE WHEN UPPER(file_status) = 'SUCCESS' THEN 1 ELSE 0 END)::BIGINT AS success_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(file_status) = 'SUCCESS' THEN tot_trnx_amt::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS success_amount,
    SUM(CASE WHEN UPPER(file_status) = 'FAILED' THEN 1 ELSE 0 END)::BIGINT AS failed_count,
    ROUND(COALESCE(SUM(CASE WHEN UPPER(file_status) = 'FAILED' THEN tot_trnx_amt::NUMERIC ELSE 0 END), 0), 0)::NUMERIC AS failed_amount
FROM common.tl_corp_bulk_file_upld_dtls
CROSS JOIN report_dates d
WHERE trunc(created_dt) between month_start and yesterday
  AND trnx_type IN ('IMPS', 'NEFT', 'RTGS')
  AND UPPER(file_status) IN ('SUCCESS', 'FAILED')
GROUP BY 1, 2

ORDER BY service, type;
